{{- /* Overrides Hugo's built-in figure so page-bundle images get resized WebP renditions. Same params as the built-in. */ -}} {{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default (.Get "caption" | markdownify | plainify) -}} {{- $img := "" -}} {{- if not (urls.Parse $src).IsAbs -}} {{- $img = or (.Page.Resources.Get $src) (resources.Get $src) -}} {{- end -}} {{- with .Get "link" }}{{ end -}} {{- if $img -}} {{- partial "img.html" (dict "image" $img "alt" $alt) -}} {{- else -}} {{- end -}} {{- if .Get "link" }}{{ end -}} {{- if or (.Get "title") (.Get "caption") (.Get "attr") -}} {{- end -}}
